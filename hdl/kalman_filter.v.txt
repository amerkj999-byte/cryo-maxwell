// kalman_filter_fixed.v — Альфа-Бета фильтр с корректным конвейером
// Q16.16: 16 бит целой части, 16 бит дробной
// ИСПРАВЛЕНО:
// 1. Обратная связь: x_est и v_est подаются на Stage 0 с задержкой 3 такта
//    (ровно столько, сколько идёт вычисление по конвейеру)
// 2. y_alpha/y_beta используют результат умножения с предыдущего такта
//    (сейчас mult_alpha уже обновлён, когда читается y_alpha)
// 3. Невязка y считается из x_pred, вычисленного на предыдущем такте
module kalman_filter_fixed (
    input  wire               clk,
    input  wire               rst_n,
    input  wire signed [31:0] position_in,   // Q16.16
    output reg  signed [31:0] position_out,  // Q16.16
    output reg  signed [31:0] velocity_out   // Q16.16
);
    // Коэффициенты Альфа-Бета фильтра в Q16.16
    parameter signed [31:0] ALPHA = 32'sh00008000;  // 0.5
    parameter signed [31:0] BETA  = 32'sh00004000;  // 0.25
    parameter signed [31:0] DT    = 32'sh00000831;  // 0.002

    // ============================================================
    // Регистры состояния фильтра (оценки с предыдущего шага)
    // ============================================================
    reg signed [31:0] x_est_reg;   // оценка позиции
    reg signed [31:0] v_est_reg;   // оценка скорости

    // ============================================================
    // Конвейер: 3 ступени
    // ============================================================

    // Stage 0: вход + предсказание позиции
    reg signed [31:0] pos_in_s0;
    reg signed [31:0] x_pred_s0;

    // Stage 1: невязка + умножение на коэффициенты
    reg signed [31:0] x_pred_s1;
    reg signed [31:0] y_s1;           // невязка
    reg signed [63:0] mult_alpha_s1;  // y * ALPHA (Q32.32)
    reg signed [63:0] mult_beta_s1;   // y * BETA  (Q32.32)

    // Stage 2: коррекция оценки
    reg signed [31:0] x_pred_s2;
    reg signed [31:0] y_alpha_s2;     // целая часть y * ALPHA
    reg signed [31:0] y_beta_s2;      // целая часть y * BETA

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            // Регистры состояния
            x_est_reg <= 32'sd0;
            v_est_reg <= 32'sd0;

            // Stage 0
            pos_in_s0  <= 32'sd0;
            x_pred_s0  <= 32'sd0;

            // Stage 1
            x_pred_s1    <= 32'sd0;
            y_s1         <= 32'sd0;
            mult_alpha_s1 <= 64'sd0;
            mult_beta_s1  <= 64'sd0;

            // Stage 2
            x_pred_s2   <= 32'sd0;
            y_alpha_s2  <= 32'sd0;
            y_beta_s2   <= 32'sd0;

            // Выходы
            position_out <= 32'sd0;
            velocity_out <= 32'sd0;
        end else begin
            // =================================================
            // Stage 0: захват входа и предсказание позиции
            // x_pred = x_est + v_est * DT
            // x_est_reg и v_est_reg — оценки с предыдущего шага
            // (обновлены 3 такта назад, что совпадает с задержкой)
            // =================================================
            pos_in_s0 <= position_in;
            x_pred_s0 <= x_est_reg + (($signed(v_est_reg) * $signed(DT)) >>> 16);

            // =================================================
            // Stage 1: невязка и умножение на ALPHA/BETA
            // y = pos_in - x_pred
            // mult_alpha = y * ALPHA  (Q16.16 × Q16.16 → Q32.32)
            // =================================================
            x_pred_s1    <= x_pred_s0;
            y_s1         <= pos_in_s0 - x_pred_s0;
            mult_alpha_s1 <= y_s1 * ALPHA;
            mult_beta_s1  <= y_s1 * BETA;

            // =================================================
            // Stage 2: извлечение целой части и коррекция
            // y_alpha = mult_alpha[47:16] — целая часть Q32.32
            // x_est = x_pred + y_alpha
            // =================================================
            x_pred_s2   <= x_pred_s1;
            y_alpha_s2  <= mult_alpha_s1[47:16];  // ← mult_alpha_s1 уже обновлён!
            y_beta_s2   <= mult_beta_s1[47:16];   // ← mult_beta_s1 уже обновлён!

            // Обновляем регистры состояния
            x_est_reg <= x_pred_s2 + y_alpha_s2;
            v_est_reg <= v_est_reg + y_beta_s2;    // скорость обновляется beta

            // Выходы
            position_out <= x_est_reg;
            velocity_out <= v_est_reg;
        end
    end
endmodule
