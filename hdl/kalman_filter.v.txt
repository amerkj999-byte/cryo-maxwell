// kalman_filter_fixed.v — Альфа-Бета фильтр с конвейером
// Q16.16: 16 бит целой части, 16 бит дробной
// Строго неблокирующие присваивания, 3-ступенчатый конвейер
module kalman_filter_fixed (
    input  wire               clk,
    input  wire               rst_n,
    input  wire signed [31:0] position_in,   // Q16.16
    output reg  signed [31:0] position_out,  // Q16.16
    output reg  signed [31:0] velocity_out   // Q16.16
);
    // Коэффициенты Альфа-Бета фильтра в Q16.16
    // Альфа = 0.5  → 32768
    // Бета  = 0.25 → 16384
    parameter signed [31:0] ALPHA = 32'sh00008000;  // 0.5
    parameter signed [31:0] BETA  = 32'sh00004000;  // 0.25
    parameter signed [31:0] DT    = 32'sh00000831;  // 0.002 (131)

    // === Конвейерные регистры ===
    // Stage 0: входные регистры (снимаем с входов)
    reg signed [31:0] pos_in_r;

    // Stage 1: предсказание и невязка
    reg signed [31:0] x_pred;        // предсказанная позиция
    reg signed [31:0] v_pred;        // предсказанная скорость
    reg signed [31:0] y;             // невязка (измерение - предсказание)

    // Stage 2: коррекция и умножение на коэффициенты
    reg signed [31:0] y_alpha;       // y * ALPHA
    reg signed [31:0] y_beta;        // y * BETA

    // Stage 3: обновление оценок
    reg signed [31:0] x_est;         // оценка позиции
    reg signed [31:0] v_est;         // оценка скорости

    // Множители (64-битные для полного произведения)
    reg signed [63:0] mult_alpha;
    reg signed [63:0] mult_beta;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            // Stage 0
            pos_in_r <= 32'sd0;

            // Stage 1
            x_pred   <= 32'sd0;
            v_pred   <= 32'sd0;
            y        <= 32'sd0;

            // Stage 2
            mult_alpha <= 64'sd0;
            mult_beta  <= 64'sd0;
            y_alpha    <= 32'sd0;
            y_beta     <= 32'sd0;

            // Stage 3
            x_est <= 32'sd0;
            v_est <= 32'sd0;

            // Выходы
            position_out <= 32'sd0;
            velocity_out <= 32'sd0;
        end else begin
            // =========== Stage 0: захват входа ===========
            pos_in_r <= position_in;

            // =========== Stage 1: предсказание ===========
            // x_pred = x_est + v_est * DT
            // Здесь v_est * DT — комбинаторно через DSP (успевает)
            x_pred <= x_est + ((v_est * DT) >>> 16);
            v_pred <= v_est;

            // Невязка: y = измерение - предсказание
            y <= pos_in_r - x_pred;

            // =========== Stage 2: умножение на коэффициенты ===========
            mult_alpha <= y * ALPHA;
            mult_beta  <= y * BETA;

            // Извлекаем Q16.16 из Q32.32 (берём старшие 32 бита)
            y_alpha <= mult_alpha[47:16];
            y_beta  <= mult_beta[47:16];

            // =========== Stage 3: коррекция ===========
            x_est <= x_pred + y_alpha;
            v_est <= v_pred + y_beta;

            // Выходы
            position_out <= x_est;
            velocity_out <= v_est;
        end
    end
endmodule
