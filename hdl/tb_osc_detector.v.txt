// tb_osc_detector.v
`timescale 1ns/1ps
module tb_osc_detector;
    reg         clk;
    reg         rst_n;
    reg  [31:0] velocity;
    wire        osc_detected;

    osc_detector dut (
        .clk_4m(clk),
        .rst_n(rst_n),
        .velocity(velocity),
        .osc_detected(osc_detected)
    );

    // Такт 4 МГц (период 250 нс)
    always #125 clk = ~clk;

    initial begin
        $dumpfile("tb_osc_detector.vcd");
        $dumpvars(0, tb_osc_detector);

        clk = 0;
        rst_n = 0;
        velocity = 32'd0;
        #500 rst_n = 1;

        // Тест 1: нет осцилляций
        #100000;
        $display("Test 1: osc_detected = %b (expected 0)", osc_detected);

        // Тест 2: осцилляции 200 Гц (4 перехода за 1 мс)
        repeat(5) begin
            velocity = 32'h3F800000; #2500; // +1 м/с
            velocity = 32'hBF800000; #2500; // -1 м/с
        end
        #50000;
        $display("Test 2: osc_detected = %b (expected 1)", osc_detected);

        // Тест 3: одиночный переход (не осцилляция)
        #100000;
        velocity = 32'hBF800000;
        #50000;
        $display("Test 3: osc_detected = %b (expected 0)", osc_detected);

        #10000 $finish;
    end
endmodule
