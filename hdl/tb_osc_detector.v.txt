// tb_osc_detector.v — тестбенч для osc_detector_ce
// ИСПРАВЛЕНО:
// 1. Тест 2 уложен в одно окно детектора (1 мс = 1 000 000 нс)
// 2. Частота колебаний подобрана так, чтобы дать 4 перехода за < 1 мс
`timescale 1ns/1ps
module tb_osc_detector;
    reg               clk_200m;
    reg               rst_n;
    reg signed [31:0] velocity;
    wire              osc_detected;

    osc_detector_ce dut (
        .clk_200m(clk_200m),
        .rst_n(rst_n),
        .velocity(velocity),
        .osc_detected(osc_detected)
    );

    // Системный такт 200 МГц (период 5 нс)
    always #2.5 clk_200m = ~clk_200m;

    // Для отладки: считаем время симуляции
    real sim_time_ns;

    initial begin
        $dumpfile("tb_osc_detector.vcd");
        $dumpvars(0, tb_osc_detector);

        clk_200m = 0;
        rst_n = 0;
        velocity = 32'sd0;
        #500 rst_n = 1;

        // Ждём инициализации
        #50000;

        // =============================================
        // Тест 1: нет осцилляций — скорость постоянна
        // =============================================
        velocity = 32'sd65536;  // +1 м/с
        #1500000;  // 1.5 мс — дольше одного окна
        sim_time_ns = $realtime / 1000.0;
        $display("[%0.f us] Test 1 (no oscillation): osc_detected = %b (expected 0)",
                 sim_time_ns, osc_detected);

        // =============================================
        // Тест 2: осцилляции 2 кГц (4 перехода за 1 мс)
        // Период 2 кГц = 500 мкс = 500 000 нс
        // Полупериод = 250 мкс = 250 000 нс
        // 2 полных цикла = 4 перехода через ноль = 1 мс
        // Всё укладывается в одно окно детектора
        // =============================================
        repeat(2) begin
            velocity =  32'sd65536;  // +1 м/с
            #250000;                 // 250 мкс
            velocity = -32'sd65536;  // -1 м/с
            #250000;                 // 250 мкс
        end
        // Ждём закрытия окна (ещё ~50 мкс с запасом)
        #100000;
        sim_time_ns = $realtime / 1000.0;
        $display("[%0.f us] Test 2 (2 kHz oscillation): osc_detected = %b (expected 1)",
                 sim_time_ns, osc_detected);

        // =============================================
        // Тест 3: одиночный переход (не осцилляция)
        // =============================================
        // Ждём нового окна
        #1200000;
        velocity = -32'sd65536;
        #100000;
        sim_time_ns = $realtime / 1000.0;
        $display("[%0.f us] Test 3 (single transition): osc_detected = %b (expected 0)",
                 sim_time_ns, osc_detected);

        // =============================================
        // Тест 4: гистерезис — малая амплитуда
        // VEL_HYST = 0.05 м/с = 3277 в Q16.16
        // Подаём колебания ±0.03 м/с = ±1966 (ниже порога)
        // =============================================
        #1200000;
        repeat(4) begin
            velocity =  32'sd1966;
            #250000;
            velocity = -32'sd1966;
            #250000;
        end
        #100000;
        sim_time_ns = $realtime / 1000.0;
        $display("[%0.f us] Test 4 (below hysteresis): osc_detected = %b (expected 0)",
                 sim_time_ns, osc_detected);

        #50000 $finish;
    end

    // Мониторинг для отладки
    always @(posedge clk_200m) begin
        if (dut.ce_4m && dut.window_timer == 0 && dut.zero_cross_count > 0) begin
            $display("[%0.f us] Window reset. Crossings in window: %d",
                     $realtime / 1000.0, dut.zero_cross_count);
        end
    end
endmodule
