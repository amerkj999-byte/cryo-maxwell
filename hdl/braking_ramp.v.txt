// braking_ramp.v — синтезируемая версия (fixed-point, без деления)
// ИСПРАВЛЕНО: сквозной конвейер с флагом диапазона
module braking_ramp (
    input  wire               clk,
    input  wire               rst_n,
    input  wire signed [31:0] velocity,    // Q16.16
    output reg  [9:0]         duty_cycle   // 0–799
);
    // Пороги скорости в Q16.16
    parameter signed [31:0] V_MAX    = 32'sh00008000; // +0.5 м/с (32768)
    parameter signed [31:0] V_MIN    = 32'sh00003333; // +0.2 м/с (13107)
    parameter DUTY_MAX = 10'd799;
    parameter DUTY_MIN = 10'd160;

    // K = (DUTY_MAX - DUTY_MIN) / (V_MAX - V_MIN) в Q16.16
    // K = 639 / 19661 * 65536 ≈ 2130
    parameter signed [31:0] K = 32'sh00000852; // 2130

    wire signed [31:0] v_abs;
    assign v_abs = (velocity[31]) ? -velocity : velocity;

    // Кодирование состояния диапазона
    parameter RANGE_BELOW = 2'd0;  // v_abs < V_MIN
    parameter RANGE_MID   = 2'd1;  // V_MIN <= v_abs <= V_MAX
    parameter RANGE_ABOVE = 2'd2;  // v_abs > V_MAX

    // === Конвейер: Stage 0 ===
    reg signed [31:0] delta_v_s0;
    reg [1:0]         range_s0;

    // === Конвейер: Stage 1 ===
    reg signed [63:0] mult_temp_s1;
    reg [1:0]         range_s1;

    // === Конвейер: Stage 2 ===
    reg signed [15:0] duty_offset_s2;
    reg [1:0]         range_s2;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            // Stage 0
            delta_v_s0 <= 32'sd0;
            range_s0   <= RANGE_BELOW;

            // Stage 1
            mult_temp_s1 <= 64'sd0;
            range_s1     <= RANGE_BELOW;

            // Stage 2
            duty_offset_s2 <= 16'sd0;
            range_s2       <= RANGE_BELOW;

            // Выход
            duty_cycle <= DUTY_MIN;
        end else begin
            // =============================================
            // Stage 0: определяем диапазон и вычисляем delta_v
            // =============================================
            if (v_abs > V_MAX) begin
                delta_v_s0 <= 32'sd0;
                range_s0   <= RANGE_ABOVE;
            end else if (v_abs < V_MIN) begin
                delta_v_s0 <= 32'sd0;
                range_s0   <= RANGE_BELOW;
            end else begin
                delta_v_s0 <= v_abs - V_MIN;
                range_s0   <= RANGE_MID;
            end

            // =============================================
            // Stage 1: умножение (только если в среднем диапазоне)
            // =============================================
            if (range_s0 == RANGE_MID)
                mult_temp_s1 <= delta_v_s0 * K;
            else
                mult_temp_s1 <= 64'sd0;

            range_s1 <= range_s0;

            // =============================================
            // Stage 2: извлечение целой части
            // =============================================
            if (range_s1 == RANGE_MID)
                duty_offset_s2 <= mult_temp_s1[47:32];  // целая часть Q32.32
            else
                duty_offset_s2 <= 16'sd0;

            range_s2 <= range_s1;

            // =============================================
            // Stage 3: финальный выбор duty_cycle
            // =============================================
            case (range_s2)
                RANGE_ABOVE: duty_cycle <= DUTY_MAX;
                RANGE_BELOW: duty_cycle <= DUTY_MIN;
                RANGE_MID:   duty_cycle <= DUTY_MIN + duty_offset_s2[9:0];
                default:     duty_cycle <= DUTY_MIN;
            endcase
        end
    end
endmodule
