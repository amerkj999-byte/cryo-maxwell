// fsm_controller.v
// FSM: Ожидание / Сжатие (КЗ) / ВМТ / Отбой (Рекуперация) / Виртуальный резистор
// ИСПРАВЛЕНО: симметричная логика через vel_abs
module fsm_controller (
    input  wire               clk,
    input  wire               rst_n,
    input  wire signed [31:0] velocity,
    input  wire signed [31:0] position,
    input  wire        [13:0] current,
    input  wire               osc_detected,
    output reg  [2:0]         mode,
    output reg                brake_en,
    output reg                rect_en,
    output reg                vr_en
);
    parameter MODE_IDLE     = 3'd0;
    parameter MODE_SHORT    = 3'd1;
    parameter MODE_TDC      = 3'd2;
    parameter MODE_RECTIFY  = 3'd3;
    parameter MODE_VR       = 3'd4;

    // Пороги в Q16.16 (знаковые)
    parameter signed [31:0] V_THRESH_UP   = 32'sh00008000;  // +0.5 м/с
    parameter signed [31:0] V_THRESH_DOWN = 32'sh00004CCD;  // +0.3 м/с
    parameter signed [31:0] TDC_POS       = 32'sd80000;     // 80 мм

    // Абсолютное значение скорости и направление
    wire signed [31:0] vel_abs;
    wire               moving_up;
    wire               moving_down;

    assign vel_abs   = (velocity[31]) ? -velocity : velocity;
    assign moving_up = (velocity[31] == 1'b0);   // velocity > 0
    assign moving_down = (velocity[31] == 1'b1); // velocity < 0

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            mode     <= MODE_IDLE;
            brake_en <= 1'b0;
            rect_en  <= 1'b0;
            vr_en    <= 1'b0;
        end else begin
            // Приоритет: осцилляции → VR
            if (osc_detected) begin
                mode     <= MODE_VR;
                brake_en <= 1'b0;
                rect_en  <= 1'b0;
                vr_en    <= 1'b1;
            end else begin
                case (mode)
                    MODE_IDLE: begin
                        brake_en <= 1'b0;
                        rect_en  <= 1'b0;
                        vr_en    <= 1'b0;

                        // Симметричная логика:
                        // Движение вверх (сжатие) → КЗ
                        if (moving_up && vel_abs > V_THRESH_UP)
                            mode <= MODE_SHORT;
                        // Движение вниз (отбой) → рекуперация
                        else if (moving_down && vel_abs > V_THRESH_DOWN)
                            mode <= MODE_RECTIFY;
                    end

                    MODE_SHORT: begin
                        brake_en <= 1'b1;
                        rect_en  <= 1'b0;
                        vr_en    <= 1'b0;
                        // Выход из КЗ по достижении ВМТ или падении тока
                        if (position >= TDC_POS && current < 14'd100)
                            mode <= MODE_TDC;
                    end

                    MODE_TDC: begin
                        brake_en <= 1'b0;
                        rect_en  <= 1'b0;
                        vr_en    <= 1'b0;
                        // Переход в рекуперацию при движении вниз
                        if (moving_down && vel_abs > V_THRESH_DOWN)
                            mode <= MODE_RECTIFY;
                    end

                    MODE_RECTIFY: begin
                        brake_en <= 1'b0;
                        rect_en  <= 1'b1;
                        vr_en    <= 1'b0;
                        // Остановка — возврат в ожидание
                        if (vel_abs == 32'sd0)
                            mode <= MODE_IDLE;
                    end

                    MODE_VR: begin
                        brake_en <= 1'b0;
                        rect_en  <= 1'b0;
                        vr_en    <= 1'b1;
                        // Выход из VR при отсутствии осцилляций и остановке
                        if (!osc_detected && vel_abs == 32'sd0)
                            mode <= MODE_IDLE;
                    end
                endcase
            end
        end
    end
endmodule
