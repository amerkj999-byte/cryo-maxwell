// can_interface.v
// Упрощённый CAN FD интерфейс (использует аппаратный CAN в STM32H743)
// ПЛИС передаёт данные через SPI → ARM → CAN
// ИСПРАВЛЕНО: передача всех 4 байт 32-битного слова последовательно
module can_interface (
    input  wire               clk,
    input  wire               rst_n,
    input  wire signed [31:0] position,
    input  wire signed [31:0] velocity,
    input  wire        [13:0] current_a,
    input  wire        [13:0] current_b,
    input  wire        [13:0] current_c,
    input  wire        [2:0]  mode,
    output wire               spi_cs,
    output wire               spi_clk,
    output wire        [7:0]  spi_mosi
);
    // Состояния передачи
    parameter IDLE       = 2'd0;
    parameter SEND_FRAME = 2'd1;

    reg [1:0] state;
    reg [2:0] frame_id;       // 0..5: 6 фреймов
    reg [1:0] byte_idx;       // 0..3: 4 байта в 32-битном слове
    reg [7:0] tx_data;
    reg       tx_valid;
    wire      tx_ready;

    // Регистр для хранения текущего передаваемого 32-битного слова
    reg [31:0] frame_data;

    // Мультиплексирование CAN-фреймов и побайтовая выдача
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state     <= IDLE;
            frame_id  <= 3'd0;
            byte_idx  <= 2'd0;
            tx_data   <= 8'd0;
            tx_valid  <= 1'b0;
            frame_data <= 32'd0;
        end else begin
            case (state)
                IDLE: begin
                    tx_valid <= 1'b0;

                    // Загружаем очередной 32-битный фрейм
                    case (frame_id)
                        3'd0: frame_data <= position;
                        3'd1: frame_data <= velocity;
                        3'd2: frame_data <= {18'd0, current_a};
                        3'd3: frame_data <= {18'd0, current_b};
                        3'd4: frame_data <= {18'd0, current_c};
                        3'd5: frame_data <= {29'd0, mode};
                    endcase

                    byte_idx <= 2'd0;
                    state    <= SEND_FRAME;
                end

                SEND_FRAME: begin
                    if (tx_ready) begin
                        // Выдаём очередной байт (от младшего к старшему)
                        case (byte_idx)
                            2'd0: tx_data <= frame_data[7:0];
                            2'd1: tx_data <= frame_data[15:8];
                            2'd2: tx_data <= frame_data[23:16];
                            2'd3: tx_data <= frame_data[31:24];
                        endcase

                        tx_valid <= 1'b1;

                        if (byte_idx == 2'd3) begin
                            // Последний байт — переход к следующему фрейму
                            frame_id <= (frame_id == 3'd5) ? 3'd0 : frame_id + 3'd1;
                            state <= IDLE;
                        end else begin
                            byte_idx <= byte_idx + 2'd1;
                        end
                    end else begin
                        tx_valid <= 1'b0;
                    end
                end
            endcase
        end
    end

    // SPI мастер
    spi_master u_spi (
        .clk(clk),
        .rst_n(rst_n),
        .tx_data(tx_data),
        .tx_valid(tx_valid),
        .tx_ready(tx_ready),
        .spi_cs(spi_cs),
        .spi_clk(spi_clk),
        .spi_mosi(spi_mosi)
    );
endmodule
