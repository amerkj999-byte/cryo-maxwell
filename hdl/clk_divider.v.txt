// clk_divider.v
// Делитель 200 МГц → 4 МГц для osc_detector
module clk_divider (
    input  wire clk_200m,
    input  wire rst_n,
    output reg  clk_4m
);
    reg [5:0] counter; // 200/4 = 50 → деление на 25 для полупериода

    always @(posedge clk_200m or negedge rst_n) begin
        if (!rst_n) begin
            counter <= 6'd0;
            clk_4m  <= 1'b0;
        end else if (counter == 6'd24) begin  // ИСПРАВЛЕНО: было 6'd49
            counter <= 6'd0;
            clk_4m  <= ~clk_4m;
        end else begin
            counter <= counter + 6'd1;
        end
    end
endmodule
