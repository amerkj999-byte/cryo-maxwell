// adc_interface.v
// Интерфейс AD9248 (14 бит, 65 MSPS, LVDS)
// ИСПРАВЛЕНО:
// 1. Преобразование Offset Binary → 2's complement
// 2. Синхронная запись в каналы без промежуточного регистра
module adc_interface (
    input  wire               clk_200m,
    input  wire               rst_n,
    input  wire               adc_clk_p,
    input  wire        [13:0] adc_data_p,
    output reg  signed [13:0] current_a,
    output reg  signed [13:0] current_b,
    output reg  signed [13:0] current_c
);
    parameter signed [13:0] OFFSET = 14'sd8192; // 0x2000

    reg [1:0] channel_sel;

    always @(posedge adc_clk_p or negedge rst_n) begin
        if (!rst_n) begin
            channel_sel <= 2'd0;
            current_a   <= 14'sd0;
            current_b   <= 14'sd0;
            current_c   <= 14'sd0;
        end else begin
            // Преобразование Offset Binary → знаковый код
            // и запись в нужный канал в одном такте
            case (channel_sel)
                2'd0: current_a <= $signed(adc_data_p) - OFFSET;
                2'd1: current_b <= $signed(adc_data_p) - OFFSET;
                2'd2: current_c <= $signed(adc_data_p) - OFFSET;
                default: ;
            endcase

            channel_sel <= (channel_sel == 2'd2) ? 2'd0 : channel_sel + 2'd1;
        end
    end
endmodule
