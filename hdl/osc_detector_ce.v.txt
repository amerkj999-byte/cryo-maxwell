// osc_detector_ce.v — Clock Enable вместо Gated Clock
// Zero-Crossing детектор паразитных осцилляций
// ИСПРАВЛЕНО:
// 1. Гистерезис проверяется на velocity_prev, а не на velocity в момент перехода
// 2. Окно 1 мс, порог 3 перехода через ноль
module osc_detector_ce (
    input  wire               clk_200m,
    input  wire               rst_n,
    input  wire signed [31:0] velocity,
    output reg                osc_detected
);
    parameter WINDOW_TICKS = 4000;   // 1 мс при CE 4 МГц
    parameter CROSS_THRESH = 3;      // порог переходов
    parameter signed [31:0] VEL_HYST = 32'sh00000CCD; // 0.05 м/с в Q16.16

    // Делитель для Clock Enable: 200 МГц / 4 МГц = 50
    parameter CE_DIVIDER = 8'd49;
    reg [7:0]  ce_counter;
    wire       ce_4m;
    assign ce_4m = (ce_counter == CE_DIVIDER);

    always @(posedge clk_200m or negedge rst_n) begin
        if (!rst_n)
            ce_counter <= 8'd0;
        else if (ce_counter == CE_DIVIDER)
            ce_counter <= 8'd0;
        else
            ce_counter <= ce_counter + 8'd1;
    end

    // Логика детектора, тактируемая ce_4m
    reg signed [31:0] velocity_prev;
    reg [7:0]         zero_cross_count;
    reg [15:0]        window_timer;

    always @(posedge clk_200m or negedge rst_n) begin
        if (!rst_n) begin
            velocity_prev    <= 32'sd0;
            zero_cross_count <= 8'd0;
            window_timer     <= 16'd0;
            osc_detected     <= 1'b0;
        end else if (ce_4m) begin
            // Детекция перехода через ноль с гистерезисом на ПРЕДЫДУЩЕМ значении
            if ((velocity_prev[31] != velocity[31]) &&
                (velocity_prev > VEL_HYST || velocity_prev < -VEL_HYST) &&
                (window_timer < WINDOW_TICKS))
                zero_cross_count <= zero_cross_count + 8'd1;

            velocity_prev <= velocity;

            // Скользящее окно
            if (window_timer >= WINDOW_TICKS) begin
                osc_detected     <= (zero_cross_count >= CROSS_THRESH);
                zero_cross_count <= 8'd0;
                window_timer     <= 16'd0;
            end else begin
                window_timer <= window_timer + 16'd1;
            end
        end
    end
endmodule
