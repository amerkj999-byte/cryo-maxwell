// deadtime_generator.v — ИСПРАВЛЕНО: асинхронное гашение
module deadtime_generator (
    input  wire         clk_200m,
    input  wire         rst_n,
    input  wire         pwm_in,
    input  wire         shutdown_async,   // ДОБАВЛЕНО
    output wire         pwm_high,
    output wire         pwm_low
);
    parameter DEADTIME_TICKS = 60;

    reg [7:0] deadtime_counter;
    reg       pwm_delayed;
    reg       pwm_high_int, pwm_low_int;

    // Асинхронное гашение — мгновенно закрывает оба ключа
    assign pwm_high = shutdown_async ? 1'b0 : pwm_high_int;
    assign pwm_low  = shutdown_async ? 1'b0 : pwm_low_int;

    always @(posedge clk_200m or negedge rst_n) begin
        if (!rst_n) begin
            pwm_high_int     <= 1'b0;
            pwm_low_int      <= 1'b0;
            pwm_delayed      <= 1'b0;
            deadtime_counter <= 8'd0;
        end else begin
            pwm_delayed <= pwm_in;

            if (pwm_in && !pwm_delayed) begin
                deadtime_counter <= DEADTIME_TICKS;
                pwm_low_int      <= 1'b0;
            end

            if (!pwm_in && pwm_delayed) begin
                deadtime_counter <= DEADTIME_TICKS;
                pwm_high_int     <= 1'b0;
            end

            if (deadtime_counter > 0)
                deadtime_counter <= deadtime_counter - 8'd1;
            else begin
                pwm_high_int <= pwm_in;
                pwm_low_int  <= ~pwm_in;
            end
        end
    end
endmodule
