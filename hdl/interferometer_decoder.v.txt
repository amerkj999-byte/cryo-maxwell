// interferometer_decoder.v
// Декодирование A/B/Z с лазерного интерферометра (1 мкм/тик)
module interferometer_decoder (
    input  wire         clk,
    input  wire         rst_n,
    input  wire         A,
    input  wire         B,
    input  wire         Z,
    output reg  [31:0]  position,
    output reg          index_pulse
);
    reg [1:0] AB_sync;
    reg [1:0] AB_prev;

    // Синхронизация асинхронных входов
    reg [1:0] A_sync, B_sync, Z_sync;

    always @(posedge clk) begin
        A_sync <= {A_sync[0], A};
        B_sync <= {B_sync[0], B};
        Z_sync <= {Z_sync[0], Z};
    end

    wire A_clean = A_sync[1];
    wire B_clean = B_sync[1];
    wire Z_clean = Z_sync[1];

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            AB_prev     <= 2'b00;
            position    <= 32'd0;
            index_pulse <= 1'b0;
        end else begin
            AB_sync <= {A_clean, B_clean};
            AB_prev <= AB_sync;

            // Декодирование Грея с немедленным инкрементом/декрементом
            // ИСПРАВЛЕНО: убран промежуточный флаг direction
            case ({AB_prev, AB_sync})
                // Переходы +1
                4'b0001: position <= position + 32'd1;
                4'b0111: position <= position + 32'd1;
                4'b1110: position <= position + 32'd1;
                4'b1000: position <= position + 32'd1;
                // Переходы -1
                4'b0010: position <= position - 32'd1;
                4'b1011: position <= position - 32'd1;
                4'b1101: position <= position - 32'd1;
                4'b0100: position <= position - 32'd1;
                default: position <= position;
            endcase

            // Индексный импульс
            if (Z_clean) begin
                position    <= 32'd0;
                index_pulse <= 1'b1;
            end else begin
                index_pulse <= 1'b0;
            end
        end
    end
endmodule
