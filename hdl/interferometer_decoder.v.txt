// interferometer_decoder.v
// Декодирование A/B/Z с лазерного интерферометра (1 мкм/тик)
// ИСПРАВЛЕНО:
// 1. Убран промежуточный флаг direction (гонка сигналов)
// 2. Сброс позиции по фронту Z, а не по уровню (устранён «липкий ноль»)
module interferometer_decoder (
    input  wire         clk,
    input  wire         rst_n,
    input  wire         A,
    input  wire         B,
    input  wire         Z,
    output reg  [31:0]  position,
    output reg          index_pulse
);
    reg [1:0] AB_sync;
    reg [1:0] AB_prev;

    // Синхронизация асинхронных входов
    reg [1:0] A_sync, B_sync, Z_sync;
    reg       Z_prev;               // ДОБАВЛЕНО: предыдущее значение Z

    always @(posedge clk) begin
        A_sync <= {A_sync[0], A};
        B_sync <= {B_sync[0], B};
        Z_sync <= {Z_sync[0], Z};
    end

    wire A_clean = A_sync[1];
    wire B_clean = B_sync[1];
    wire Z_clean = Z_sync[1];

    // Детектор фронта Z: 0→1
    wire Z_rising;
    assign Z_rising = Z_clean && !Z_prev;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            AB_prev     <= 2'b00;
            Z_prev      <= 1'b0;        // ДОБАВЛЕНО
            position    <= 32'd0;
            index_pulse <= 1'b0;
        end else begin
            AB_sync <= {A_clean, B_clean};
            AB_prev <= AB_sync;
            Z_prev  <= Z_clean;         // ДОБАВЛЕНО: запоминаем Z

            // Декодирование Грея — инкремент/декремент
            case ({AB_prev, AB_sync})
                // Переходы +1
                4'b0001: position <= position + 32'd1;
                4'b0111: position <= position + 32'd1;
                4'b1110: position <= position + 32'd1;
                4'b1000: position <= position + 32'd1;
                // Переходы -1
                4'b0010: position <= position - 32'd1;
                4'b1011: position <= position - 32'd1;
                4'b1101: position <= position - 32'd1;
                4'b0100: position <= position - 32'd1;
                default: position <= position;
            endcase

            // Сброс позиции ТОЛЬКО по фронту Z (0→1)
            // После сброса position продолжает считать шаги A/B,
            // даже если Z_clean всё ещё = 1
            if (Z_rising) begin
                position    <= 32'd0;
                index_pulse <= 1'b1;
            end else begin
                index_pulse <= 1'b0;
            end
        end
    end
endmodule
