// top.v — Cryo-Maxwell V3.0 Active Damper Controller
// Финальная версия со всеми исправлениями:
// 1. signed-порты для Q16.16 (fsm_controller, osc_detector_ce, pi_controller)
// 2. LVDS-буферы для АЦП (adc_clk_n, adc_data_n)
// 3. Асинхронное гашение ключей через shutdown_async
// 4. Clock Enable вместо Gated Clock (osc_detector_ce)
// 5. Убран clk_divider
// 6. heartbeat_led на счётчике
// 7. Три независимых ПИ-регулятора (фазы A, B, C)
// 8. Три раздельных канала ШИМ (pwm_muxed_a/b/c) с синхронным обновлением
// 9. Имя модуля фильтра: kalman_filter_fixed
// 10. SPI: spi_miso проброшен в can_interface, подтяжка PULLUP
// 11. ШИМ-мультиплексор тактированный (убираем комбинаторный skew)
module top (
    // Системные
    input  wire         clk_200m_p,
    input  wire         clk_200m_n,
    input  wire         rst_n,

    // Интерферометр
    input  wire         inter_A,
    input  wire         inter_B,
    input  wire         inter_Z,

    // АЦП (LVDS) — полные дифференциальные пары
    input  wire         adc_clk_p,
    input  wire         adc_clk_n,
    input  wire [13:0]  adc_data_p,
    input  wire [13:0]  adc_data_n,

    // Драйверы ключей (3 фазы × 2 ключа)
    output wire         phase_A_high,
    output wire         phase_A_low,
    output wire         phase_B_high,
    output wire         phase_B_low,
    output wire         phase_C_high,
    output wire         phase_C_low,

    // CAN/SPI интерфейс
    output wire         spi_cs,
    output wire         spi_clk,
    output wire         spi_mosi,
    (* PULLTYPE = "PULLUP" *) input wire spi_miso,  // внутренняя подтяжка
    output wire         fault_led,
    output wire         heartbeat_led
);

    // =================================================================
    // Внутренние сигналы
    // =================================================================
    wire                clk_200m;
    wire signed [31:0]  position_raw;
    wire signed [31:0]  position_filt;
    wire signed [31:0]  velocity_filt;
    wire                index_pulse;
    wire signed [13:0]  current_a, current_b, current_c;
    wire                osc_detected;
    wire        [9:0]   duty_cycle;
    wire        [2:0]   mode;
    wire                brake_en, rect_en, vr_en;
    wire        [9:0]   pwm_duty_a;
    wire        [9:0]   pwm_duty_b;
    wire        [9:0]   pwm_duty_c;
    wire                fault, shutdown;
    wire                shutdown_async;
    wire                phase_A_raw, phase_B_raw, phase_C_raw;

    // Сигналы после LVDS-буферов
    wire                adc_clk_se;
    wire        [13:0]  adc_data_se;

    // =================================================================
    // Дифференциальный буфер системного такта
    // =================================================================
    IBUFGDS #(.DIFF_TERM("TRUE")) clk_buf (
        .I(clk_200m_p),
        .IB(clk_200m_n),
        .O(clk_200m)
    );

    // =================================================================
    // Дифференциальные буферы АЦП
    // =================================================================
    IBUFGDS #(.DIFF_TERM("TRUE")) adc_clk_buf (
        .I(adc_clk_p),
        .IB(adc_clk_n),
        .O(adc_clk_se)
    );

    genvar i;
    generate
        for (i = 0; i < 14; i = i + 1) begin : adc_data_buf
            IBUFDS #(.DIFF_TERM("TRUE")) adc_data_buf_inst (
                .I(adc_data_p[i]),
                .IB(adc_data_n[i]),
                .O(adc_data_se[i])
            );
        end
    endgenerate

    // =================================================================
    // Декодер интерферометра (сброс позиции по фронту Z)
    // =================================================================
    interferometer_decoder u_inter (
        .clk(clk_200m),
        .rst_n(rst_n),
        .A(inter_A),
        .B(inter_B),
        .Z(inter_Z),
        .position(position_raw),
        .index_pulse(index_pulse)
    );

    // =================================================================
    // Альфа-Бета фильтр (4-ступенчатый конвейер, Q16.16)
    // =================================================================
    kalman_filter_fixed u_kalman (
        .clk(clk_200m),
        .rst_n(rst_n),
        .position_in(position_raw),
        .position_out(position_filt),
        .velocity_out(velocity_filt)
    );

    // =================================================================
    // Интерфейс АЦП (Offset Binary → signed, без промежуточного регистра)
    // =================================================================
    adc_interface u_adc (
        .clk_200m(clk_200m),
        .rst_n(rst_n),
        .adc_clk_p(adc_clk_se),
        .adc_data_p(adc_data_se),
        .current_a(current_a),
        .current_b(current_b),
        .current_c(current_c)
    );

    // =================================================================
    // Детектор осцилляций (Clock Enable, гистерезис на velocity_prev)
    // =================================================================
    osc_detector_ce u_osc (
        .clk_200m(clk_200m),
        .rst_n(rst_n),
        .velocity(velocity_filt),
        .osc_detected(osc_detected)
    );

    // =================================================================
    // Генератор рампы торможения (конвейер с флагом диапазона)
    // =================================================================
    braking_ramp u_braking (
        .clk(clk_200m),
        .rst_n(rst_n),
        .velocity(velocity_filt),
        .duty_cycle(duty_cycle)
    );

    // =================================================================
    // Конечный автомат (симметричная логика, выход из MODE_SHORT по отбою)
    // =================================================================
    fsm_controller u_fsm (
        .clk(clk_200m),
        .rst_n(rst_n),
        .velocity(velocity_filt),
        .position(position_filt),
        .current(current_a),
        .osc_detected(osc_detected),
        .mode(mode),
        .brake_en(brake_en),
        .rect_en(rect_en),
        .vr_en(vr_en)
    );

    // =================================================================
    // PI-регуляторы — по одному на каждую фазу (4-ступенчатый конвейер)
    // =================================================================
    pi_controller u_pi_a (
        .clk(clk_200m),
        .rst_n(rst_n),
        .current_in(current_a),
        .target_current(14'sd5000),
        .pwm_duty(pwm_duty_a)
    );

    pi_controller u_pi_b (
        .clk(clk_200m),
        .rst_n(rst_n),
        .current_in(current_b),
        .target_current(14'sd5000),
        .pwm_duty(pwm_duty_b)
    );

    pi_controller u_pi_c (
        .clk(clk_200m),
        .rst_n(rst_n),
        .current_in(current_c),
        .target_current(14'sd5000),
        .pwm_duty(pwm_duty_c)
    );

    // =================================================================
    // Мультиплексор ШИМ — тактированный (убирает комбинаторный skew)
    // =================================================================
    reg [9:0] pwm_muxed_a, pwm_muxed_b, pwm_muxed_c;

    always @(posedge clk_200m or negedge rst_n) begin
        if (!rst_n) begin
            pwm_muxed_a <= 10'd0;
            pwm_muxed_b <= 10'd0;
            pwm_muxed_c <= 10'd0;
        end else begin
            if (shutdown) begin
                pwm_muxed_a <= 10'd0;
                pwm_muxed_b <= 10'd0;
                pwm_muxed_c <= 10'd0;
            end else if (brake_en) begin
                pwm_muxed_a <= duty_cycle;
                pwm_muxed_b <= duty_cycle;
                pwm_muxed_c <= duty_cycle;
            end else if (vr_en) begin
                pwm_muxed_a <= pwm_duty_a;
                pwm_muxed_b <= pwm_duty_b;
                pwm_muxed_c <= pwm_duty_c;
            end else if (rect_en) begin
                pwm_muxed_a <= 10'd0;
                pwm_muxed_b <= 10'd0;
                pwm_muxed_c <= 10'd0;
            end else begin
                pwm_muxed_a <= 10'd0;
                pwm_muxed_b <= 10'd0;
                pwm_muxed_c <= 10'd0;
            end
        end
    end

    // =================================================================
    // ШИМ-генератор (три канала, синхронное обновление по началу периода)
    // =================================================================
    pwm_module u_pwm (
        .pwm_clk(clk_200m),
        .rst_n(rst_n),
        .duty_a(pwm_muxed_a),
        .duty_b(pwm_muxed_b),
        .duty_c(pwm_muxed_c),
        .pwm_a(phase_A_raw),
        .pwm_b(phase_B_raw),
        .pwm_c(phase_C_raw)
    );

    // =================================================================
    // Мёртвое время + асинхронное гашение
    // =================================================================
    deadtime_generator u_dt_a (
        .clk_200m(clk_200m),
        .rst_n(rst_n),
        .pwm_in(phase_A_raw),
        .shutdown_async(shutdown_async),
        .pwm_high(phase_A_high),
        .pwm_low(phase_A_low)
    );

    deadtime_generator u_dt_b (
        .clk_200m(clk_200m),
        .rst_n(rst_n),
        .pwm_in(phase_B_raw),
        .shutdown_async(shutdown_async),
        .pwm_high(phase_B_high),
        .pwm_low(phase_B_low)
    );

    deadtime_generator u_dt_c (
        .clk_200m(clk_200m),
        .rst_n(rst_n),
        .pwm_in(phase_C_raw),
        .shutdown_async(shutdown_async),
        .pwm_high(phase_C_high),
        .pwm_low(phase_C_low)
    );

    // =================================================================
    // Защита по току (signed, двунаправленная)
    // =================================================================
    overcurrent_protect u_protect (
        .clk(clk_200m),
        .rst_n(rst_n),
        .current_a(current_a),
        .current_b(current_b),
        .current_c(current_c),
        .fault(fault),
        .shutdown(shutdown),
        .shutdown_async(shutdown_async)
    );

    // =================================================================
    // CAN/SPI интерфейс (spi_miso проброшен)
    // =================================================================
    can_interface u_can (
        .clk(clk_200m),
        .rst_n(rst_n),
        .position(position_filt),
        .velocity(velocity_filt),
        .current_a(current_a),
        .current_b(current_b),
        .current_c(current_c),
        .mode(mode),
        .spi_miso(spi_miso),
        .spi_cs(spi_cs),
        .spi_clk(spi_clk),
        .spi_mosi(spi_mosi)
    );

    // =================================================================
    // Индикация
    // =================================================================
    assign fault_led = fault;

    reg [25:0] heartbeat_cnt;
    always @(posedge clk_200m or negedge rst_n) begin
        if (!rst_n)
            heartbeat_cnt <= 26'd0;
        else
            heartbeat_cnt <= heartbeat_cnt + 26'd1;
    end
    assign heartbeat_led = heartbeat_cnt[25];

endmodule
