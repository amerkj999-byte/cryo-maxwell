// overcurrent_protect.v — защита по току (полностью signed, двунаправленная)
// ИСПРАВЛЕНО:
// 1. Порты signed
// 2. Константа I_MAX — signed, в пределах 14-битного знакового диапазона
// 3. Двунаправленная защита: > +I_MAX и < -I_MAX
module overcurrent_protect (
    input  wire               clk,
    input  wire               rst_n,
    input  wire signed [13:0] current_a,
    input  wire signed [13:0] current_b,
    input  wire signed [13:0] current_c,
    output reg                fault,
    output reg                shutdown,
    output wire               shutdown_async
);
    // Знаковый порог: 15 А в отсчётах АЦП
    // Калибруется под датчик тока и шунт
    // Максимум знакового 14-бит: +8191
    // Пример: 1 А = 400 отсчётов → 15 А = 6000
    parameter signed [13:0] I_MAX = 14'sd6000;

    // Двунаправленный асинхронный компаратор
    // Оба операнда signed — Verilog выполняет знаковое сравнение
    assign shutdown_async = (current_a > I_MAX) || (current_a < -I_MAX) ||
                            (current_b > I_MAX) || (current_b < -I_MAX) ||
                            (current_c > I_MAX) || (current_c < -I_MAX);

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            fault    <= 1'b0;
            shutdown <= 1'b0;
        end else begin
            fault <= shutdown_async;
            if (shutdown_async)
                shutdown <= 1'b1;  // защёлка до аппаратного сброса
        end
    end
endmodule
