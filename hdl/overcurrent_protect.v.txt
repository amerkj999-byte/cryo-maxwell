// overcurrent_protect.v — ИСПРАВЛЕНО: добавлен асинхронный выход
module overcurrent_protect (
    input  wire         clk,
    input  wire         rst_n,
    input  wire [13:0]  current_a,
    input  wire [13:0]  current_b,
    input  wire [13:0]  current_c,
    output reg          fault,
    output reg          shutdown,
    output wire         shutdown_async   // ДОБАВЛЕНО: комбинаторный
);
    parameter I_MAX = 14'd15000;

    // Асинхронный компаратор — срабатывает мгновенно
    assign shutdown_async = (current_a > I_MAX) ||
                            (current_b > I_MAX) ||
                            (current_c > I_MAX);

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            fault    <= 1'b0;
            shutdown <= 1'b0;
        end else begin
            fault    <= shutdown_async;
            if (shutdown_async)
                shutdown <= 1'b1;
        end
    end
endmodule
