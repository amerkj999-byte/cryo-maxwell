// pi_controller.v — ИСПРАВЛЕНО (защита от усечения)
module pi_controller (
    input  wire         clk,
    input  wire         rst_n,
    input  wire [13:0]  current_in,
    input  wire [13:0]  target_current,
    output reg  [9:0]   pwm_duty       // 10 бит (0–799)
);
    parameter Kp = 8'h20;
    parameter Ki = 8'h04;

    reg [31:0] integral;
    reg [31:0] error;
    reg [31:0] pwm_tmp;    // ДОБАВЛЕНО: временный регистр

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            integral <= 32'd0;
            pwm_duty <= 10'd0;
            pwm_tmp  <= 32'd0;       // ДОБАВЛЕНО
        end else begin
            error = target_current - current_in;

            // Антивиндовка
            if (pwm_duty < 10'd799 && pwm_duty > 10'd0)
                integral <= integral + Ki * error;

            // Считаем во временный регистр
            pwm_tmp <= Kp * error + integral;

            // Ограничение на 32-битном уровне
            if (pwm_tmp > 32'd799)
                pwm_tmp <= 32'd799;

            // Присваиваем младшие 10 бит
            pwm_duty <= pwm_tmp[9:0];
        end
    end
endmodule
