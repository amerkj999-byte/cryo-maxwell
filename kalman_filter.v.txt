// kalman_filter_fixed.v — Fixed-point реализация
// Q16.16: 16 бит целой части, 16 бит дробной
module kalman_filter_fixed (
    input  wire         clk,
    input  wire         rst_n,
    input  wire [31:0]  position_in,   // Q16.16
    output reg  [31:0]  position_out,  // Q16.16
    output reg  [31:0]  velocity_out   // Q16.16
);
    // Константы в Q16.16
    parameter P_INIT = 32'h00010000; // 1.0
    parameter Q_VAL  = 32'h00028F5C; // 0.01
    parameter R_VAL  = 32'h00019999; // 0.1
    parameter DT     = 32'h00000831; // 0.002

    reg [31:0] x_est, v_est;
    reg [31:0] p_xx, p_xv, p_vv;
    reg [63:0] mult_temp; // Для умножений 32×32=64
    reg [31:0] y, S, K_x, K_v;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            x_est <= 32'd0;
            v_est <= 32'd0;
            p_xx  <= P_INIT;
            p_xv  <= 32'd0;
            p_vv  <= P_INIT;
        end else begin
            // Предсказание
            // x = x + v*dt
            mult_temp = v_est * DT;
            x_est <= x_est + mult_temp[47:16]; // старшие 32 бита из 64
            // p_xx = p_xx + 2*p_xv*dt + p_vv*dt*dt + Q
            mult_temp = p_xv * DT;
            p_xx <= p_xx + {mult_temp[46:16], 1'b0} + Q_VAL; // упрощённо
            p_xv <= p_xv + p_vv;
            p_vv <= p_vv + Q_VAL;

            // Обновление
            y = position_in - x_est;
            S = p_xx + R_VAL;

            // K = p_xx / S (приближённо, через сдвиг)
            // Для fixed-point: K = (p_xx << 16) / S — не делаем,
            // используем табличное приближение или IP-ядро деления
            // Здесь заглушка с константой
            K_x = 32'h00008000; // 0.5 в Q16.16
            K_v = 32'h00004000; // 0.25

            x_est <= x_est + ((K_x * y) >> 16);
            v_est <= v_est + ((K_v * y) >> 16);

            position_out <= x_est;
            velocity_out <= v_est;
        end
    end
endmodule
