// pwm_module.v — ИСПРАВЛЕНО
module pwm_module (
    input  wire         pwm_clk,    // 200 МГц
    input  wire         rst_n,
    input  wire [9:0]   duty_a,     // 0–799 (10 бит)
    input  wire [9:0]   duty_b,
    input  wire [9:0]   duty_c,
    output reg          pwm_a,
    output reg          pwm_b,
    output reg          pwm_c
);
    reg [9:0] counter;
    parameter MAX_CNT = 10'd799;

    always @(posedge pwm_clk or negedge rst_n) begin
        if (!rst_n) begin
            counter <= 10'd0;
            pwm_a   <= 1'b0;
            pwm_b   <= 1'b0;
            pwm_c   <= 1'b0;
        end else begin
            counter <= (counter == MAX_CNT) ? 10'd0 : counter + 10'd1;
            pwm_a   <= (counter < duty_a);
            pwm_b   <= (counter < duty_b);
            pwm_c   <= (counter < duty_c);
        end
    end
endmodule
