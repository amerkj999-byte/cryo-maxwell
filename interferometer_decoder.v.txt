// interferometer_decoder.v
// Декодирование A/B/Z с лазерного интерферометра (1 мкм/тик)
module interferometer_decoder (
    input  wire         clk,
    input  wire         rst_n,
    input  wire         A,
    input  wire         B,
    input  wire         Z,
    output reg  [31:0]  position,
    output reg          index_pulse
);
    reg [1:0] AB_sync;
    reg [1:0] AB_prev;
    reg       direction;

    // Синхронизация асинхронных входов (метастабильность)
    reg [1:0] A_sync, B_sync, Z_sync;

    always @(posedge clk) begin
        A_sync <= {A_sync[0], A};
        B_sync <= {B_sync[0], B};
        Z_sync <= {Z_sync[0], Z};
    end

    wire A_clean = A_sync[1];
    wire B_clean = B_sync[1];
    wire Z_clean = Z_sync[1];

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            AB_prev     <= 2'b00;
            position    <= 32'd0;
            index_pulse <= 1'b0;
        end else begin
            AB_sync <= {A_clean, B_clean};
            AB_prev <= AB_sync;

            // Детекция направления по Грею
            case ({AB_prev, AB_sync})
                4'b0001: direction = 1'b1;  // +1
                4'b0111: direction = 1'b1;
                4'b1110: direction = 1'b1;
                4'b1000: direction = 1'b1;
                4'b0010: direction = 1'b0;  // -1
                4'b1011: direction = 1'b0;
                4'b1101: direction = 1'b0;
                4'b0100: direction = 1'b0;
                default: direction = 1'b0;
            endcase

            if (AB_prev != AB_sync) begin
                if (direction)
                    position <= position + 32'd1;
                else
                    position <= position - 32'd1;
            end

            // Индексный импульс (Z) — сброс позиции
            if (Z_clean) begin
                position    <= 32'd0;
                index_pulse <= 1'b1;
            end else begin
                index_pulse <= 1'b0;
            end
        end
    end
endmodule
