// can_interface.v
// Упрощённый CAN FD интерфейс (использует аппаратный CAN в STM32H743)
// ПЛИС передаёт данные через SPI → ARM → CAN
module can_interface (
    input  wire         clk,
    input  wire         rst_n,
    input  wire [31:0]  position,
    input  wire [31:0]  velocity,
    input  wire [13:0]  current_a,
    input  wire [13:0]  current_b,
    input  wire [13:0]  current_c,
    input  wire [2:0]   mode,
    output wire         spi_cs,
    output wire         spi_clk,
    output wire [7:0]   spi_mosi
);
    // SPI-передатчик (режим 0)
    reg [7:0]  tx_data;
    reg        tx_valid;
    wire       tx_ready;

    // Мультиплексирование CAN-фреймов
    reg [2:0]  frame_id;
    reg [31:0] frame_data;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            frame_id   <= 3'd0;
            frame_data <= 32'd0;
            tx_valid   <= 1'b0;
        end else begin
            // Циклическая передача фреймов
            case (frame_id)
                3'd0: frame_data <= position;
                3'd1: frame_data <= velocity;
                3'd2: frame_data <= {18'd0, current_a};
                3'd3: frame_data <= {18'd0, current_b};
                3'd4: frame_data <= {18'd0, current_c};
                3'd5: frame_data <= {29'd0, mode};
            endcase

            tx_data <= frame_data[7:0]; // передача младшего байта
            tx_valid <= 1'b1;

            if (tx_ready) begin
                frame_id <= (frame_id == 3'd5) ? 3'd0 : frame_id + 3'd1;
                tx_valid <= 1'b0;
            end
        end
    end

    // Простой SPI мастер
    spi_master u_spi (
        .clk(clk),
        .rst_n(rst_n),
        .tx_data(tx_data),
        .tx_valid(tx_valid),
        .tx_ready(tx_ready),
        .spi_cs(spi_cs),
        .spi_clk(spi_clk),
        .spi_mosi(spi_mosi)
    );
endmodule
